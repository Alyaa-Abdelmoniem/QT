cmake_minimum_required(VERSION 3.16)

project(SmartHome VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Quick)

qt_standard_project_setup()

qt_add_executable(appSmartHome
    main.cpp
    translator.h
    translator.cpp
)

qt_add_qml_module(appSmartHome
    URI SmartHome
    VERSION 1.0

    QML_FILES
        Main.qml
        LoginPage.qml
        DashboardPage.qml
        SettingsPage.qml

    RESOURCES
        images/home.png
        images/light.png
        images/fan.png
        images/ac.png
        images/garage.png
        images/user.png
)

target_link_libraries(appSmartHome
    PRIVATE Qt6::Quick
)
