cmake_minimum_required(VERSION 3.16)

project(ECommerceApp VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2)

qt_standard_project_setup(REQUIRES 6.10)

qt_add_executable(appECommerceApp
    main.cpp
)

qt_add_qml_module(appECommerceApp
    URI ecommerce

    QML_FILES
        Main.qml
        SplashScreen.qml
        HomePage.qml

    RESOURCES
        images/logo.png
        images/banner.png
        images/Home.png
        images/Phones.png
        images/Computers.png
        images/Fashion.png
        images/Shopping.png
        images/Cars.png
)

target_link_libraries(appECommerceApp
    PRIVATE
        Qt6::Quick
        Qt6::QuickControls2
)

set_target_properties(appECommerceApp PROPERTIES
    WIN32_EXECUTABLE TRUE
)
